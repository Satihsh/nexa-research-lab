"""Faithful transliteration of NEXA_Risk.mqh PearsonCorrelation (lines 174-204) on SYNTHETIC data.
closeA/closeB are indexed by shift (index 0 = forming bar); a missing bar is iClose()==0."""
import numpy as np

def pearson_mql(closeA, closeB, barsBack=100):
    retA, retB, valid = np.zeros(barsBack), np.zeros(barsBack), 0
    for i in range(barsBack):
        a0, a1, b0, b1 = closeA[i+1], closeA[i+2], closeB[i+1], closeB[i+2]
        if a1 <= 0 or b1 <= 0 or a0 <= 0 or b0 <= 0:
            continue                                    # retA[i]=retB[i]=0 (zero-filled, still counted below)
        retA[i], retB[i] = (a0-a1)/a1, (b0-b1)/b1; valid += 1
    if valid < 3: return 0.0
    ma, mb = retA.mean(), retB.mean()                   # divides by barsBack, not valid
    cov = ((retA-ma)*(retB-mb)).sum(); va = ((retA-ma)**2).sum(); vb = ((retB-mb)**2).sum()
    if va <= 0 or vb <= 0: return 0.0
    return cov/np.sqrt(va*vb)

rng = np.random.default_rng(0); n = 400
common = rng.normal(0, 1e-4, n); A_r = common + rng.normal(0, 4e-5, n); B_r = common + rng.normal(0, 4e-5, n)
A = 1.10*np.exp(np.cumsum(A_r[::-1]))[::-1]; B = 1.27*np.exp(np.cumsum(B_r[::-1]))[::-1]   # shift-indexed
base = pearson_mql(A, B); print(f"(0) aligned, full history          corr={base:.3f}  -> {'BLOCK' if abs(base)>=0.75 else 'PASS'}")
Z = np.zeros_like(B); c = pearson_mql(A, Z); print(f"(a) B history absent (iClose=0)     corr={c:.3f}  -> {'BLOCK' if abs(c)>=0.75 else 'PASS'}")
for frac in (0.1, 0.2, 0.3):
    Bm = B.copy(); idx = rng.choice(np.arange(1, 102), int(frac*101), replace=False); Bm[idx] = 0
    c = pearson_mql(A, Bm); print(f"(b) {int(frac*100)}% of B bars missing (=0)     corr={c:.3f}  -> {'BLOCK' if abs(c)>=0.75 else 'PASS'}")
for k in (1, 3, 12, 60):
    Bs = np.r_[B[k:], np.full(k, B[-1])]               # B lagged by k bars at the same shift (gap in B, e.g. XAU break)
    c = pearson_mql(A, Bs); print(f"(c) B misaligned by {k:>2} bars (shift)  corr={c:.3f}  -> {'BLOCK' if abs(c)>=0.75 else 'PASS'}")
print("partial misalignment: bars 1..m aligned, older bars offset by a 60-bar gap in B (e.g. XAU daily break)")
for m in (80, 60, 40, 20, 5):
    Bp = np.r_[B[:m+1], B[m+1+60:]]; Bp = np.r_[Bp, np.full(len(B)-len(Bp), Bp[-1])]
    c = pearson_mql(A, Bp); print(f"   aligned recent bars m={m:>2}/101  corr={c:.3f} -> {'BLOCK' if abs(c)>=0.75 else 'PASS'}")
