def tree_sha(d):
    h = hashlib.sha256()
    for root, _, files in sorted(os.walk(d)):
        for f in sorted(files):
            p = os.path.join(root, f)
            h.update(p.encode()); h.update(open(p, "rb").read())
    return h.hexdigest()
